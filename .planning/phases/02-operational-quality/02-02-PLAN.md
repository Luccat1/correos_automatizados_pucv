```javascript
/**
 * Server-side function called from SendDialog.html to fetch pre-flight data.
 * Returns quota, sender alias, batch warning threshold, and processable row count.
 * @returns {Object|null}
 */
function getPreflightData() {
  var resolved = _getSelectedTemplate();
  if (!resolved) return null;

  var config = resolved.config;
  var template = resolved.template;
  var sheet = _getDataSheet(config.dataSheet);
  if (!sheet) return null;

  var headers = mapHeaders(sheet);
  var statusColName = template.statusColumn || 'Estado_Envio';
  var statusIdx = headers[statusColName.trim().toLowerCase()];

  // Count processable rows (empty status or ERROR status)
  var lastRow = sheet.getLastRow();
  var processableCount = 0;
  var totalDataRows = lastRow - 1;

  if (totalDataRows > 0 && statusIdx !== undefined) {
    var statusValues = sheet.getRange(2, statusIdx + 1, totalDataRows, 1).getValues();
    for (var r = 0; r < statusValues.length; r++) {
      var st = String(statusValues[r][0] || '').toLowerCase();
      if (!st || st.includes('error') || st.includes('saltado')) {
        processableCount++;
      }
    }
  } else {
    // No status column yet means all rows are processable
    processableCount = totalDataRows;
  }

  return {
    quota: MailApp.getRemainingDailyQuota(),
    senderAlias: config.senderAlias || template.senderAlias || '',
    batchThreshold: parseInt(config.extras.batch_warning_threshold) || 50,
    processableRows: processableCount,
    totalRows: totalDataRows
  };
}
```

**4. Update the JavaScript** (replace the entire `<script>` block with):

```html
<script>
  var preflightData = null;
  var batchWarningActive = false;

  // Fetch pre-flight data on dialog load (ROBUST-04, UX-02)
  google.script.run
    .withSuccessHandler(function(data) {
      if (!data) return;
      preflightData = data;

      // Update quota display
      document.getElementById('quota-count').textContent = data.quota;

      // Update processable row count
      document.getElementById('processable-count').textContent = data.processableRows + ' de ' + data.totalRows;

      // Update sender alias if server has more info
      if (data.senderAlias) {
        document.getElementById('sender-alias').textContent = data.senderAlias;
      }

      // Show batch warning if processable rows exceed threshold (ROBUST-05)
      if (data.processableRows > data.batchThreshold) {
        batchWarningActive = true;
        document.getElementById('warning-count').textContent = data.processableRows;
        document.getElementById('warning-threshold').textContent = data.batchThreshold;
        document.getElementById('batch-warning').style.display = 'block';
        document.getElementById('submitBtn').disabled = true;
      }
    })
    .withFailureHandler(function(err) {
      document.getElementById('quota-count').textContent = 'error';
      document.getElementById('processable-count').textContent = 'error';
      console.error('getPreflightData failed:', err);
    })
    .getPreflightData();

  // Large-batch checkbox enables/disables submit (ROBUST-05)
  document.getElementById('confirm-large-batch').addEventListener('change', function() {
    if (batchWarningActive) {
      document.getElementById('submitBtn').disabled = !this.checked;
    }
  });

  // Form submission
  document.getElementById('sendForm').addEventListener('submit', function(e) {
    e.preventDefault();

    var submitBtn = document.getElementById('submitBtn');
    var loader = document.getElementById('loader');

    submitBtn.disabled = true;
    loader.style.display = 'block';

    var formData = {
      subject: document.getElementById('subject').value,
      extraNote: document.getElementById('extraNote').value,
      cc: document.getElementById('cc').value,
      bcc: document.getElementById('bcc').value,
      asDraft: document.querySelector('input[name="asDraft"]').value === 'true'
    };

    google.script.run
      .withSuccessHandler(function() {
        google.script.host.close();
      })
      .withFailureHandler(function(err) {
        alert('Error: ' + err);
        submitBtn.disabled = false;
        loader.style.display = 'none';
      })
      .executeBatchFromDialog(formData);
  });
</script>
```

Key behaviors:
- Quota and processable count load asynchronously via `getPreflightData()` on dialog open
- If processable rows > threshold: red warning box appears, submit button disabled until checkbox is checked
- If processable rows <= threshold: no warning, submit button stays enabled
- Sender alias shown from server-rendered template variable (immediate) and optionally updated by async data
